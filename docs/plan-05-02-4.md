**任务一：正式网格实验（3 seeds × 3 budgets = 9 组）**

**目标**：为论文 Table 2（主结果表）生成跨 seed 均值 ± 标准差数据。

**实验配置**：
- Seeds：41, 42, 43
- Budgets：1.65, 1.70, 1.75
- 每组运行所有 6 个调度策略：random, round_robin, aoi, custom_ppo, dqn, cmdp_dqn
- 每组 custom_ppo 训练步数：100k（与单 seed 验证一致）
- 输出目录：`reports/v2_forecast_eval_grid/budget{budget}_seed{seed}/evaluation/`

**新建脚本**：`scripts/26_v2_grid_experiment.py`

```python
"""
正式网格实验：3 seeds × 3 budgets = 9 组
每组运行所有 6 个调度策略，输出 forecast_weighted_mae_overall 及分层指标
"""
import argparse
import subprocess
import os
from itertools import product

SEEDS = [41, 42, 43]
BUDGETS = [1.65, 1.70, 1.75]
POLICIES = ["random", "round_robin", "aoi", "custom_ppo", "dqn", "cmdp_dqn"]
BASE_OUTPUT_DIR = "reports/v2_forecast_eval_grid"

def run_single(budget: float, seed: int):
    """运行单组实验（一个 budget × seed 组合的所有策略）"""
    output_dir = f"{BASE_OUTPUT_DIR}/budget{str(budget).replace('.', 'p')}_seed{seed}"
    os.makedirs(output_dir, exist_ok=True)
    
    # Step 1: 训练 custom_ppo（若 checkpoint 不存在）
    ckpt_path = f"{output_dir}/custom_ppo_checkpoint.pt"
    if not os.path.exists(ckpt_path):
        subprocess.run([
            "python", "scripts/25_v2_train_custom_ppo.py",
            "--budget", str(budget),
            "--seed", str(seed),
            "--total-timesteps", "100000",
            "--output-dir", output_dir,
            "--checkpoint-path", ckpt_path,
        ], check=True)
    
    # Step 2: 运行所有策略的 rollout + 评估
    subprocess.run([
        "python", "scripts/24_v2_evaluate_rollouts.py",
        "--budget", str(budget),
        "--seed", str(seed),
        "--output-dir", output_dir,
        "--custom-ppo-checkpoint", ckpt_path,
    ], check=True)

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--dry-run", action="store_true", help="只打印命令，不执行")
    parser.add_argument("--budget", type=float, default=None, help="只运行指定 budget（调试用）")
    parser.add_argument("--seed", type=int, default=None, help="只运行指定 seed（调试用）")
    args = parser.parse_args()
    
    budgets = [args.budget] if args.budget else BUDGETS
    seeds = [args.seed] if args.seed else SEEDS
    
    for budget, seed in product(budgets, seeds):
        print(f"Running budget={budget}, seed={seed}...")
        if not args.dry_run:
            run_single(budget, seed)
    
    print("Grid experiment complete.")
```

**任务二：汇总脚本（生成论文图表数据）**

**新建脚本**：`scripts/27_v2_aggregate_results.py`

**功能**：读取 9 组实验结果，计算跨 seed 均值 ± 标准差，输出论文所需 CSV 文件。

```python
"""
汇总 9 组网格实验结果，生成论文图表数据
输出：
  reports/v2_paper_tables/
    table2_main_results.csv        # Table 2：主结果表（均值 ± 标准差）
    table3_by_variable.csv         # Table 3a：逐变量指标
    table3_by_condition.csv        # Table 3b：分层指标（event / low_temp）
    figure4_learning_curves/       # Figure 4：学习曲线数据（每 seed 一个 CSV）
"""
import pandas as pd
import numpy as np
import os
from itertools import product

SEEDS = [41, 42, 43]
BUDGETS = [1.65, 1.70, 1.75]
POLICIES = ["random", "round_robin", "aoi", "custom_ppo", "dqn", "cmdp_dqn",
            "full_open_unconstrained", "feasible_static_projected"]
BASE_INPUT_DIR = "reports/v2_forecast_eval_grid"
BASE_OUTPUT_DIR = "reports/v2_paper_tables"

def load_overall_results():
    """加载所有组的主指标结果"""
    records = []
    for budget, seed in product(BUDGETS, SEEDS):
        budget_str = str(budget).replace('.', 'p')
        csv_path = f"{BASE_INPUT_DIR}/budget{budget_str}_seed{seed}/evaluation/v2_eval_overall.csv"
        if not os.path.exists(csv_path):
            print(f"WARNING: Missing {csv_path}")
            continue
        df = pd.read_csv(csv_path)
        df["budget"] = budget
        df["seed"] = seed
        records.append(df)
    return pd.concat(records, ignore_index=True)

def generate_table2(df_all):
    """
    Table 2：主结果表
    行：policy（6 个调度策略 + 2 个 oracle/上界）
    列：budget=1.65 / 1.70 / 1.75 的 forecast_weighted_mae_overall（均值 ± 标准差）
    格式：mean ± std（跨 3 seeds）
    """
    results = []
    for budget in BUDGETS:
        df_b = df_all[df_all["budget"] == budget]
        for policy in POLICIES:
            df_p = df_b[df_b["policy"] == policy]["forecast_weighted_mae_overall"]
            if len(df_p) == 0:
                continue
            mean_val = df_p.mean()
            std_val = df_p.std(ddof=1) if len(df_p) > 1 else 0.0
            results.append({
                "policy": policy,
                "budget": budget,
                "mean": round(mean_val, 4),
                "std": round(std_val, 4),
                "mean_pm_std": f"{mean_val:.4f} ± {std_val:.4f}",
            })
    df_table2 = pd.DataFrame(results)
    # 透视为宽表（行=policy，列=budget）
    df_pivot = df_table2.pivot_table(
        index="policy", columns="budget", values="mean_pm_std", aggfunc="first"
    )
    # 按 budget=1.70 的均值排序（升序，越小越好）
    order_col = 1.70
    df_order = df_table2[df_table2["budget"] == order_col].set_index("policy")["mean"]
    df_pivot = df_pivot.loc[df_order.sort_values().index]
    return df_pivot

def generate_table3_by_variable(df_all, budget=1.70):
    """
    Table 3a：逐变量指标（budget=1.70，跨 3 seeds 均值）
    行：policy，列：9 个预测变量的 forecast_mae
    """
    records = []
    for seed in SEEDS:
        budget_str = str(budget).replace('.', 'p')
        csv_path = f"{BASE_INPUT_DIR}/budget{budget_str}_seed{seed}/evaluation/v2_eval_by_variable.csv"
        if not os.path.exists(csv_path):
            print(f"WARNING: Missing {csv_path}")
            continue
        df = pd.read_csv(csv_path)
        df["seed"] = seed
        records.append(df)
    if not records:
        return pd.DataFrame()
    df_var = pd.concat(records, ignore_index=True)
    # 按 policy 和 variable 分组，计算均值
    df_agg = df_var.groupby(["policy", "variable"])["forecast_mae"].mean().reset_index()
    df_pivot = df_agg.pivot_table(index="policy", columns="variable", values="forecast_mae")
    return df_pivot

def generate_table3_by_condition(df_all, budget=1.70):
    """
    Table 3b：分层指标（budget=1.70，跨 3 seeds 均值 ± 标准差）
    行：policy，列：overall / event（wind>8 m/s）/ low_temp（T<-30°C）
    """
    records = []
    for seed in SEEDS:
        budget_str = str(budget).replace('.', 'p')
        csv_path = f"{BASE_INPUT_DIR}/budget{budget_str}_seed{seed}/evaluation/v2_eval_by_condition.csv"
        if not os.path.exists(csv_path):
            print(f"WARNING: Missing {csv_path}")
            continue
        df = pd.read_csv(csv_path)
        df["seed"] = seed
        records.append(df)
    if not records:
        return pd.DataFrame()
    df_cond = pd.concat(records, ignore_index=True)
    # 按 policy 和 condition 分组
    df_agg = df_cond.groupby(["policy", "condition"])["forecast_weighted_mae"].agg(
        mean="mean", std=lambda x: x.std(ddof=1) if len(x) > 1 else 0.0
    ).reset_index()
    df_agg["mean_pm_std"] = df_agg.apply(
        lambda r: f"{r['mean']:.4f} ± {r['std']:.4f}", axis=1
    )
    df_pivot = df_agg.pivot_table(
        index="policy", columns="condition", values="mean_pm_std", aggfunc="first"
    )
    return df_pivot

def generate_figure4_data():
    """
    Figure 4：学习曲线数据
    输出每个 seed 的 custom_ppo 训练过程中 forecast_weighted_mae 随步数的变化
    （从 custom_ppo 训练日志中读取，若存在）
    """
    output_dir = f"{BASE_OUTPUT_DIR}/figure4_learning_curves"
    os.makedirs(output_dir, exist_ok=True)
    for seed in SEEDS:
        budget_str = "1p70"
        log_path = f"{BASE_INPUT_DIR}/budget{budget_str}_seed{seed}/custom_ppo_training_log.csv"
        if os.path.exists(log_path):
            import shutil
            shutil.copy(log_path, f"{output_dir}/seed{seed}_training_log.csv")
            print(f"Copied training log for seed={seed}")
        else:
            print(f"WARNING: No training log found for seed={seed} at {log_path}")

if __name__ == "__main__":
    os.makedirs(BASE_OUTPUT_DIR, exist_ok=True)
    
    # 加载所有结果
    df_all = load_overall_results()
    print(f"Loaded {len(df_all)} records from {len(df_all.groupby(['budget', 'seed']))} groups")
    
    # 生成 Table 2
    df_table2 = generate_table2(df_all)
    df_table2.to_csv(f"{BASE_OUTPUT_DIR}/table2_main_results.csv")
    print("Table 2 saved.")
    print(df_table2.to_string())
    
    # 生成 Table 3a（逐变量，budget=1.70）
    df_table3a = generate_table3_by_variable(df_all, budget=1.70)
    if not df_table3a.empty:
        df_table3a.to_csv(f"{BASE_OUTPUT_DIR}/table3_by_variable.csv")
        print("Table 3a saved.")
    
    # 生成 Table 3b（分层，budget=1.70）
    df_table3b = generate_table3_by_condition(df_all, budget=1.70)
    if not df_table3b.empty:
        df_table3b.to_csv(f"{BASE_OUTPUT_DIR}/table3_by_condition.csv")
        print("Table 3b saved.")
    
    # 生成 Figure 4 数据
    generate_figure4_data()
    
    print(f"\nAll paper table data saved to {BASE_OUTPUT_DIR}/")
```
